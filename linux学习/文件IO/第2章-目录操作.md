# 第2章 目录操作

> 本章整理自 `3.标准IO二.pdf` 第 8–19 页（内容对应原始 `1.目录操作.pdf`）。返回 [[00-文件IO目录|📑 文件IO目录]]。

## 本章要点

- 目录也是文件，存储的是目录项；对目录的操作即逐个读取目录项。
- `opendir / readdir / closedir` 遍历目录；`struct dirent` 中仅 `d_ino`、`d_name` 为全 Linux 通用成员，可移植代码只用这两个。
- `mkdir` 创建目录；`stat / fstat / lstat` 获取文件属性，`struct stat` 的 `st_mode` 用 `S_IFMT` 判断文件类型、`S_IRUSR` 等宏判断权限。
- `unlink` 删除文件、`rmdir` 删除空目录、`remove` 两者皆可；`truncate / ftruncate` 把文件截短至指定长度。
- `getcwd` 获取进程当前工作路径；`chdir` 改变的是进程（而非终端）的工作路径。
- 作业：仿写 `ls -a/-l`；仿写 `cp` 目录拷贝——章末给出 fileCopy 与递归 directoryCopy 完整示例。

## 目录操作

在 Linux 下对于目录其实也是一个文件。但是目录不能直接进行读写因为目录中所存储的是目录项数据

**目录项**：

- 其实就是记录了该目录下面的子目录和文件的一些信息
- 文件名字，文件大小 ....

对目录进行操作，无非就是去读取目录项来一个一个的获取该目录下的文件夹和文件。

#### 读取目录的内容

就是读取一个已经打开的目录的目录项

```c
READDIR (3)                                                     Linux Programmer 's
Manual                                                   READDIR(3)

NAME
        readdir - read a directory

SYNOPSIS
        #include <dirent.h>

        struct dirent *readdir (DIR *dirp );
         /*
             @ 描述：
                  在打开的目录文件中，读取目录文件的内容
             @dirp ：
                  目录文件的目录指针
             @return
                  成功返回目录文件中的一个子目录 / 文件的结构体指针
                  失败 / 读取完毕返回 NULL

             注意：每次读取目录都会往后自动偏移
         */

//  目录项结构体
struct dirent {
    //  子目录 / 文件的 inode 编号
    ino_t d_ino ;        /* Inode number */

    //  子目录 / 文件在目录文件的中偏移量
    off_t d_off ;        /* Not an offset; see below */

    //  结构体的大小
    unsigned short d_reclen ;     /* Length of this record */

    //  子目录 / 文件的类型
    unsigned char d_type ;       /* Type of file; not supported by all filesystem
types */

    //  子目录 / 文件的名称（不包含路径）
    char d_name [256 ];  /* Null-terminated filename */
};
```

**注意：该结构体成员变量：是所有 Linux 系统都支持的。如果想让代码具有更好的可移植性，该结构体代码中，只能用** `d_ino 、 d_name` **这两个成员变量。**

```c
    //  文件所有者用户的 ID
    uid_t st_uid ;          /* User ID of owner */

    //  文件所属用户组的 ID
    gid_t st_gid ;          /* Group ID of owner */

    //  设备号，（如果这个文件的是一个设备文件的话）
    dev_t st_rdev ;         /* Device ID (if special file) */

    //  文件的内容的大小
    off_t st_size ;         /* Total size, in bytes */
    /*

    blksize_t st_blksize ;      /* Block size for filesystem I/O */

    struct timespec st_mtim ;   /* Time of last modification */
```

作业：

- `请仿写 ls 命令的-a 和-l 的功能`
- `请仿写 cp 命令的目录拷贝功能`

```c
#include <sys/types.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <unistd.h>
#include <dirent.h>
#include <stdio.h>
#include <string.h>

void fileCopy(char *sourceFile,char *destFile)
{
    //  打开两个文件
    int AfileId = open(sourceFile,O_RDONLY);//  被拷贝的文件，需要修改内容吗？ x
    int BfileId = open(destFile,O_WRONLY | O_CREAT,0777);//  拷贝文件，需要读取内容？ x

    char buf[2]={0};
    //  循环
    int realCount = 0;
    do
    {
        realCount = read(AfileId,buf,2);
        write(BfileId,buf,realCount);
    }
    while(realCount == 2);

    close(AfileId);
    close(BfileId);
}

/*
    目录的拷贝
         目录项：记录子文件和子文件夹的信息
         读取目录项
         将读取的到的目录项，复制到指定位置
*/
void directoryCopy(const char *srcDir,const char *destDir)
{
    /*
         辅助变量
    */
    char absWorkPath[1024]   = {0}; //  当前的程序的工作路径的绝对路径
    char absSourcePath[1024] = {0}; //  源文件夹的绝对路径
    char absDestPath[1024]   = {0}; //  目标路径的绝对路径

    //  记录当前程序的工作路径
    getcwd(absWorkPath,1024);

    //  获取源文件夹的路径
    chdir(srcDir);  //  切换至源文件夹的路径下
    getcwd(absSourcePath,1024);

    //  切换路径到目标文件夹下
    /*
        1. 目标路径不存在
        2. 目标路径存在
    */
    //  先切换回工作路径
    chdir(absWorkPath);

    //  考虑创建目标路径
    mkdir(destDir,0777); //  如果文件夹存在，则不会重复创建，直接返回

    //  切换目标路径下
    chdir(destDir);

    //  获取目标路径的绝对路径
    getcwd(absDestPath,1024);
    //------------------------------ 读取源路径下的目录项 ----------------------------
-
    DIR *sourceDirPoint = opendir(absSourcePath);
    struct dirent *direntPoint =  NULL;
    struct stat    fileStat;

    while(direntPoint = readdir(sourceDirPoint))
    {
        //  切换到源路径
        chdir(absSourcePath);

        //  跳过当前目录和上一级目录的目录项
        if(!strcmp(direntPoint->d_name,".")||!strcmp(direntPoint->d_name,".."))
            continue;

        //  获取文件的属性
        stat(direntPoint->d_name,&fileStat);

        //  拼接路径
        char childSourcePath[2048]={0}; //  源路径下面的子目录 完整路径
        sprintf(childSourcePath,"%s/%s",absSourcePath,direntPoint->d_name);

        char childDestPath[2048]={0};//  目录路径下面的子目录 完整路径
        sprintf(childDestPath,"%s/%s",absDestPath,direntPoint->d_name);

        //  判断是目录还是文件
        if(S_ISDIR(fileStat.st_mode))
        {
            printf(" 目录： %s\n",direntPoint->d_name);
            //  递归去进行子目录的拷贝
            directoryCopy(childSourcePath,childDestPath);
        }
        else
        {
            printf(" 文件： %s\n",direntPoint->d_name);
            fileCopy(childSourcePath,childDestPath);
        }
    }
    closedir(sourceDirPoint);
}

int main(int argc,const char *argv[])
{
    if(argc != 3)
    {
        printf("%s <source directory> <dest directory>\n",argv[0]);
        return -1;
    }

    //  进行目录拷贝
    directoryCopy(argv[1],argv[2]);

    return 0;
}
```
